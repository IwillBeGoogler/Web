export interface Drawable {
  draw(): void;
}

export interface Repository<T extends { id: number }> extends Drawable {
  readonly name: string;
  find(id: string): Promise<T | undefined>;
  save(item: T): Promise<void>;
}

export abstract class Shape implements Drawable {
  private static count = 0;
  protected readonly id: number;
  public label?: string;
  #secret: string = 'x';
  constructor(private readonly repo: Repository<Item>, id: number) {
    this.id = id;
  }
  abstract area(): number;
  static create(kind: string): Shape {
    throw new Error(kind);
  }
  get title(): string {
    return '';
  }
  protected describe(prefix = '', ...rest: string[]): string {
    return prefix;
  }
  draw(): void {}
}

export class Circle extends Shape {
  radius: number = 1;
  area(): number {
    return Math.PI * this.radius ** 2;
  }
}
