T = TypeVar("T")


class Drawable(Protocol):
    def draw(self) -> None: ...


class Repository(ABC, Generic[T]):

    class Fake(Base):  # inside a docstring

        self.__secret: str = "x"
        self.items: List[Item] = []
        self.owner: Optional[Owner] = None

    def area(self) -> float:
        pass

    @classmethod
    def make(cls, a: int): ...

    @property
    def title(self) -> str: ...

    def _helper(self): ...

    def __private(self, *args, **kw): ...
