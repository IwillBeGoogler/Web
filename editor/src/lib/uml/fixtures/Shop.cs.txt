using System;
using System.Collections.Generic;

namespace Shop.Model
{
    public interface IRepository<T>
    {
        T Find(int id);
        void Save(T item);
        IEnumerable<T> All { get; }
    }

    public interface IPricer : IDisposable
    {
        decimal Price(Order order);
    }

    public enum Status
    {
        Draft = 0,
        Paid,
        Shipped
    }

    public abstract class Entity
    {
        public int Id { get; protected set; }
        protected DateTime CreatedAt;
        public abstract string Describe();
    }

    [Serializable]
    public class Customer : Entity, IComparable<Customer>
    {
        private readonly List<Order> _orders = new();
        public string Name { get; set; } = "";
        public string? Email { get; init; }
        public Address Address { get; set; }
        public IReadOnlyList<Order> Orders => _orders;
        public static int Count;
        public const string Kind = "customer";

        public Customer(string name) { Name = name; }

        public override string Describe() => Name;
        public void AddOrder(Order order, int priority = 0) { _orders.Add(order); }
        public static Customer Create(string name, params string[] tags) { return new Customer(name); }
        public int CompareTo(Customer other) => 0;
        private int Secret() { return 1; }
    }

    public record Address(string Street, string City);

    public struct Money
    {
        public decimal Amount { get; set; }
        public string Currency { get; set; }
    }

    public class Order : Entity
    {
        public Customer Customer { get; set; }
        public List<OrderLine> Lines { get; } = new List<OrderLine>();
        public Dictionary<string, Money> Fees { get; set; }
        public Status Status { get; set; }
        public override string Describe() { return "order"; }

        public class OrderLine
        {
            public string Sku { get; set; }
        }
    }
}
